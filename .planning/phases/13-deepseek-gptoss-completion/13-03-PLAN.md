    test_deepseek() {
        local model_name="DeepSeek-Coder-V2-Lite-Instruct-4bit-mlx"
        local port=8082
        
        echo "Testing DeepSeek-Coder-V2-Lite..."
        
        # 1. Test model discovery
        echo "  - Checking model directory..."
        if [[ ! -d "./models/${model_name}" ]]; then
            echo "    ⚠️ Model not found, skipping DeepSeek tests"
            return 0
        fi
        
        # 2. Test weight loading
        echo "  - Testing weight loading..."
        timeout 60 ./zig-out/bin/zlx --model "./models/${model_name}" --port ${port} &
        local server_pid=$!
        sleep 5
        
        # 3. Test /v1/models endpoint
        echo "  - Testing /v1/models..."
        local models_response=$(curl -s http://localhost:${port}/v1/models)
        if [[ ! $models_response == *"${model_name}"* ]]; then
            echo "    ❌ Model not listed in /v1/models"
            kill $server_pid 2>/dev/null
            return 1
        fi
        echo "    ✓ Model listed"
        
        # 4. Test chat completion
        echo "  - Testing chat completion..."
        local response=$(curl -s -X POST http://localhost:${port}/v1/chat/completions \
            -H "Content-Type: application/json" \
            -d "{\"model\": \"${model_name}\", \"messages\": [{\"role\": \"user\", \"content\": \"Hello\"}], \"max_tokens\": 10}")
        
        if [[ ! $response == *"choices"* ]]; then
            echo "    ❌ No choices in response"
            echo "    Response: ${response}"
            kill $server_pid 2>/dev/null
            return 1
        fi
        echo "    ✓ Chat completion works"
        
        # 5. Test TurboQuant compression
        echo "  - Testing TurboQuant compression..."
        local compression_info=$(curl -s http://localhost:${port}/v1/metrics | grep -o '"kv_compression_ratio":[0-9.]*' | cut -d: -f2)
        if [[ -n $compression_info && $(echo "$compression_info > 4.0" | bc -l) -eq 1 ]]; then
            echo "    ✓ TurboQuant active (${compression_info}x compression)"
        else
            echo "    ⚠️ TurboQuant not detected or compression low"
        fi
        
        # 6. Test memory usage
        echo "  - Checking memory usage..."
        local mem_usage=$(get_memory_usage $server_pid)
        if [[ $(echo "$mem_usage < 16384" | bc -l) -eq 1 ]]; then
            echo "    ✓ Memory under 16GB (${mem_usage}MB)"
        else
            echo "    ❌ Memory over 16GB (${mem_usage}MB)"
        fi
        
        kill $server_pid 2>/dev/null
        wait $server_pid 2>/dev/null
        
        echo "  ✓ DeepSeek tests passed"
        return 0
    }

    download_gptoss_model() {
        local model_id="mlx-community/gpt-oss-20b-MXFP4-Q4"
        local target_dir="./models/gpt-oss-20b-MXFP4-Q4"
        
        echo "Downloading GPT-OSS from HuggingFace..."
        
        # Create directory
        mkdir -p "${target_dir}"
        
        # Download config.json
        curl -L -o "${target_dir}/config.json" \
            "https://huggingface.co/${model_id}/raw/main/config.json"
        
        # Download tokenizer files
        curl -L -o "${target_dir}/tokenizer.json" \
            "https://huggingface.co/${model_id}/resolve/main/tokenizer.json"
        
        # Download safetensors index
        curl -L -o "${target_dir}/model.safetensors.index.json" \
            "https://huggingface.co/${model_id}/raw/main/model.safetensors.index.json"
        
        # Download weight shards (this will take a while)
        local shard_count=$(jq '.weight_map | values | group_by(.) | length' "${target_dir}/model.safetensors.index.json")
        echo "Downloading ${shard_count} weight shards (~11GB total)..."
        
        for i in $(seq 1 $shard_count); do
            local shard_file=$(printf "model-%05d-of-%05d.safetensors" $i $shard_count)
            echo "  Downloading ${shard_file}..."
            
            # Use resume capability
            curl -C - -L -o "${target_dir}/${shard_file}" \
                "https://huggingface.co/${model_id}/resolve/main/${shard_file}" || {
                echo "    ❌ Failed to download ${shard_file}"
                return 1
            }
        done
        
        echo "✓ GPT-OSS download complete"
        return 0
    }

    const std = @import("std");
    const testing = std.testing;
    const loader = @import("inference/loader.zig");
    const deepseek = @import("deepseek.zig");
    const gpt_oss = @import("gpt_oss.zig");
    
    test "DeepSeek weight loading and dequantization" {
        const allocator = testing.allocator;
        
        // Skip if model not present
        const model_path = "./models/DeepSeek-Coder-V2-Lite-Instruct-4bit-mlx";
        if (!try std.fs.cwd().access(model_path, .{})) {
            std.log.info("DeepSeek model not found, skipping test", .{});
            return;
        }
        
        // 1. Detect model type
        const config_info = try loader.loadConfig(allocator, model_path);
        const model_type = loader.detectModelType(config_info);
        try testing.expectEqual(.deepseek, model_type);
        
        // 2. Load weights (includes dequantization)
        const weights = try loader.loadModelWeights(allocator, model_type, config_info);
        defer loader.freeModelWeights(allocator, weights);
        
        // 3. Verify weights are valid (not empty)
        switch (weights) {
            .deepseek => |dw| {
                // Check token embedding
                try testing.expect(!mlx.arrayIsEmpty(dw.token_embedding));
                
                // Check first layer
                try testing.expect(dw.layers.len > 0);
                try testing.expect(!mlx.arrayIsEmpty(dw.layers[0].input_layernorm));
                
                // Check that quantized weights were dequantized
                // (they should have float dtype, not packed int)
                const first_attn_weight = dw.layers[0].self_attn.q_proj;
                const dtype = mlx.arrayDtype(first_attn_weight);
                try testing.expect(dtype == .f16 or dtype == .f32);
            },
            else => return error.WrongModelType,
        }
        
        std.log.info("✓ DeepSeek weight loading and dequantization test passed", .{});
    }
    
    test "GPT-OSS weight loading" {
        const allocator = testing.allocator;
        
        // Skip if model not present
        const model_path = "./models/gpt-oss-20b-MXFP4-Q4";
        if (!try std.fs.cwd().access(model_path, .{})) {
            std.log.info("GPT-OSS model not found, skipping test", .{});
            return;
        }
        
        // 1. Detect model type
        const config_info = try loader.loadConfig(allocator, model_path);
        const model_type = loader.detectModelType(config_info);
        try testing.expectEqual(.gpt_oss, model_type);
        
        // 2. Load weights
        const weights = try loader.loadModelWeights(allocator, model_type, config_info);

        // 3. Verify weights

            .gpt_oss => |gw| {
                try testing.expect(!mlx.arrayIsEmpty(gw.token_embedding));
                try testing.expect(gw.layers.len == 24);
            },
            else => return error.WrongModelType,
        }
        
        std.log.info("✓ GPT-OSS weight loading test passed", .{});
    }
