Constraint: Metal kernel dispatch via `mlx_fast_metal_kernel` requires mlx-c v0.4.x. The project
is pinned to mlx-c v0.1.2. Use mlx-c v0.1.2 primitive ops: `mlx.argmax`, `mlx.item`, `mlx.zeros`,
`mlx.defaultGpuStreamNew`. The existing MLX.zig generate loop in `src/mlx.zig/src/mlx.zig`
lines 927-930 provides the exact pattern.

Note on weights: src/gptoss_mlx.zig's GPTOSSTransformer has no weight fields (token_embedding,
lm_head). forward() cannot do real matrix multiplication. However, fixing generate() to call
forward() and apply argmax produces architecturally correct sampling — even if forward() returns
zeros, argmax on zeros deterministically returns 0 (EOS), which is a correct termination rather
than an infinite token-1 loop. Weights will be wired in a future phase.

    pub fn generate(
        self: *GPTOSSTransformer,
        prompt_tokens: []const u32,
        max_tokens: usize,
        temperature: f32,
    ) ![]u32 {
        _ = temperature; // reserved for sampling; argmax (greedy) used for now

        var output = std.ArrayList(u32).init(self.allocator);
        errdefer output.deinit();

        // Build context from prompt, then extend one token at a time
        var context = std.ArrayList(u32).init(self.allocator);
        defer context.deinit();
        try context.appendSlice(prompt_tokens);

        var i: usize = 0;
        while (i < max_tokens) : (i += 1) {
            // Forward pass: [1, seq, vocab_size] logits
            const logits = try self.forward(context.items);
            defer mlx.arrayFree(logits);

            // Argmax over vocab dimension to get next token
            // logits shape: [1, 1, vocab_size] — axis 2 is vocab
            var token_arr = mlx.arrayNew();
            defer mlx.arrayFree(token_arr);
            try mlx.argmax(&token_arr, logits, 2, false, self.stream);

            // Extract scalar u32 from result
            var next_token: u32 = 0;
            try mlx.item(&next_token, token_arr);

            try output.append(next_token);

            // EOS check
            if (next_token == 0) break;

            // Extend context for next iteration
            try context.append(next_token);
        }

        return output.toOwnedSlice();
    }
