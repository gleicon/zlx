```zig
    allocator: std.mem.Allocator,
    gptoss_backend: *MLXGPTOSSBackend,

    pub fn init(allocator: std.mem.Allocator, gptoss: *MLXGPTOSSBackend) ChatGPTOSSHandler;

```

CRITICAL: `ChatGPTOSSHandler.handle` is NOT a free function. It cannot be called as
`chat_gptoss.ChatGPTOSSHandler.handle(req, res)`. It requires an initialized instance.
The wrapper in server.zig must hold a module-level `ChatGPTOSSHandler` and call
`g_chat_gptoss_handler.?.handle(req, res)`.

  <name>Task 2: Dispatch /v1/chat/completions to ChatGPTOSSHandler instance for gpt-oss models</name>
  <files>src/api/server.zig</files>
  <action>
    Modify the existing `handleChatCompletions` free function in `server.zig`. Currently:
    ```zig
    fn handleChatCompletions(req: *httpz.Request, res: *httpz.Response) !void {
        try handlers.handleChatCompletions(req, res);
    }
    ```

    Replace with a version that peeks at the `model` field and dispatches to the
    `g_chat_gptoss_handler` instance when the model name starts with `"gpt-oss"`:

    ```zig
    fn handleChatCompletions(req: *httpz.Request, res: *httpz.Response) !void {
        // Peek at model field to dispatch GPT-OSS requests to dedicated handler
        const body = req.body() orelse {
            try handlers.handleChatCompletions(req, res);
            return;
        };

        const ModelPeek = struct { model: []const u8 = "" };
        const peek = std.json.parseFromSlice(ModelPeek, req.arena, body, .{ .ignore_unknown_fields = true }) catch {
            // Parse failed — fall through to generic handler which will return proper error
            try handlers.handleChatCompletions(req, res);
            return;
        };
        defer peek.deinit();

        if (std.mem.startsWith(u8, peek.value.model, "gpt-oss") or
            std.mem.startsWith(u8, peek.value.model, "gptoss"))
        {
            if (g_chat_gptoss_handler) |*handler| {
                // ChatGPTOSSHandler.handle is an instance method — must call on *handler
                try handler.handle(req, res);
            } else {
                res.status = 503;
                try res.json(.{ .@"error" = "GPT-OSS handler not initialized" }, .{});
            }
        } else {
            try handlers.handleChatCompletions(req, res);
        }
    }
    ```

    Note on `req.arena`: this is the per-request arena allocator provided by httpz zig-0.13.
    It is freed automatically after the request completes. Use it for the temporary ModelPeek
    parse to avoid leaks.

    Note on `isGptOssModel`: The check mirrors the `isGptOssModel` helper in chat_gptoss.zig
    (accepts both `"gpt-oss"` and `"gptoss"` prefixes). Keep them consistent.

    Do not modify any other handler functions or routes.
