    1. Create `Pcg32Rng` struct at module level:
    ```zig
    /// PCG32 deterministic random number generator
    /// Same seed produces same sequence — required for OpenAI API compatibility
    pub const Pcg32Rng = struct {
        state: u64,
        inc: u64,
        
        const MULTIPLIER: u64 = 6364136223846793005;
        const DEFAULT_INC: u64 = 1442695040888963407;
        
        pub fn init(seed: u32) Pcg32Rng {
            var rng = Pcg32Rng{
                .state = 0,
                .inc = (DEFAULT_INC << 1) | 1,
            };
            _ = rng.next(); // Initialize state
            rng.state = rng.state +% seed;
            _ = rng.next();
            return rng;
        }
        
        fn next(self: *Pcg32Rng) u32 {
            const old_state = self.state;
            self.state = old_state *% MULTIPLIER +% self.inc;
            const xor_shifted: u32 = @intCast(((old_state >> 18) ^ old_state) >> 27);
            const rot: u32 = @intCast(old_state >> 59);
            return (xor_shifted >> rot) | (xor_shifted << ((~rot +% 1) & 31));
        }
        
        /// Return random float in range [0, 1)
        pub fn random(self: *Pcg32Rng) f32 {
            return @as(f32, @floatFromInt(self.next())) / @as(f32, @floatFromInt(@as(u32, @intCast(0xFFFFFFFF))));
        }
    };
    ```

    **Ensure temperature=0 selects highest probability token (greedy):**
    
    The current implementation in `generator.zig:next()` already has this logic:
    ```zig
    if (self.options.temperature == 0) {
        // Greedy: pick the highest probability token
        var max_prob: f32 = 0;
        var max_idx: u32 = 0;
        for (0..@intCast(vocab_size)) |i| {
            const p = probs_data[i];
            if (p > max_prob) {
                max_prob = p;
                max_idx = @intCast(i);
            }
        }
        next_token = max_idx;
    }
    ```
    
    However, per D-18, we need to ensure this happens BEFORE the softmax, not after.
    Current code applies temperature scaling, then softmax, then greedy selection from probabilities.
    
    **Correct implementation per D-18:**
    1. If temperature == 0: apply argmax directly to logits (before temperature/softmax)
    2. This gives true greedy selection without numerical issues from probability scaling
    
    Update the code flow in `next()`:
    ```zig
    // Take logits for last position
    var last_logits = mlx.arrayNew();
    defer mlx.arrayFree(last_logits);
    try mlx.take(&last_logits, self.logits_array, mlx.int(-1), 1, stream);
    
    if (self.options.temperature == 0) {
        // D-18: Greedy selection — argmax on raw logits
        var next_token_arr = mlx.arrayNew();
        defer mlx.arrayFree(next_token_arr);
        try mlx.argmax(&next_token_arr, last_logits, 1, false, stream);
        try mlx.item(&next_token, next_token_arr);
    } else {
        // Apply temperature scaling
        var scaled_logits = mlx.arrayNew();
        defer mlx.arrayFree(scaled_logits);
        
        if (self.options.temperature != 1.0) {
            const temp_scalar = mlx.float(self.options.temperature);
            try mlx.divide(&scaled_logits, last_logits, temp_scalar, stream);
        } else {
            try mlx.arraySet(&scaled_logits, last_logits);
        }
        
        // Apply softmax and sample
        // ... existing code ...
    }
    ```
    
    This change ensures:
    - temperature=0 bypasses temperature scaling entirely
    - True argmax on logits = true greedy selection
    - No numerical issues from division by zero or tiny probabilities
