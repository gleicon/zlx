       ```zig
       for (self.layers, 0..) |*layer, i| {
       ```

       ```zig
       fn rmsNorm(self: DeepSeekTransformer, x: mlx.Array, weight: mlx.Array, eps: f32) !mlx.Array {
           _ = self;
           const squared = try mlx.square(x);
           const mean = try mlx.mean(squared, .{ .axis = -1, .keep_dims = true });
           const eps_array = mlx.array(eps);
           const denominator = try mlx.sqrt(try mlx.add(mean, eps_array));
           const normalized = try mlx.divide(x, denominator);
           return try mlx.multiply(normalized, weight);
       }
       ```

  <name>Task 3: Implement autoregressive generation loop</name>
  <files>src/mlx.zig/src/deepseek.zig</files>
  <behavior>
    - Initialize compressed KV caches for all 27 layers
    - Loop max_tokens times
    - Forward pass with current token(s)
    - Sample next token with temperature
    - Update caches
    - Return generated token sequence
  </behavior>
  <action>
    Implement generate() method:
    
    1. Initialize compressed KV caches:
       ```zig
       var caches = try self.allocator.alloc(*mla.CompressedKVCache, self.config.num_hidden_layers);
       defer self.allocator.free(caches);
       
       for (0..self.config.num_hidden_layers) |i| {
           caches[i] = try self.allocator.create(mla.CompressedKVCache);
           caches[i].* = try mla.CompressedKVCache.init(
               self.allocator,
               self.config.max_position_embeddings,
               self.config.latent_dim,
           );
       }
       defer {
           for (caches) |cache| {
               cache.deinit();
               self.allocator.destroy(cache);
           }
       }
       ```
    
    2. Generation loop:
       ```zig
       var current_ids = prompt_ids;
       var output_tokens = std.ArrayList(i32).init(self.allocator);
       defer output_tokens.deinit();
       
       for (0..max_tokens) |_| {
           // Forward pass
           const logits = try self.forward(current_ids, caches);
           defer mlx.arrayFree(logits);
           
           // Get last token logits
           const next_token_logits = try logits.index(-1);
           defer mlx.arrayFree(next_token_logits);
           
           // Sample with temperature
           const next_token = try self.sample(next_token_logits, temperature);
           
           // Append to output
           try output_tokens.append(next_token);
           
           // Prepare for next iteration (single token)
           mlx.arrayFree(current_ids);
           current_ids = mlx.array(next_token);
       }
       ```
    
    3. Implement sample() with temperature:
       ```zig
       fn sample(self: DeepSeekTransformer, logits: mlx.Array, temperature: f32) !i32 {
           _ = self;
           const scaled = try mlx.divide(logits, mlx.array(temperature));
           defer mlx.arrayFree(scaled);
           const probs = try mlx.softmax(scaled, .{ .axis = -1 });
           defer mlx.arrayFree(probs);
           const token = try mlx.argmax(probs, .{ .axis = -1 });
           defer mlx.arrayFree(token);
           return try mlx.arrayItem(token);
       }
       ```
    
    4. Return generated sequence as MLX array
  </action>
