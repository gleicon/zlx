    1. Define routeOnCpu():
       ```zig
       fn routeOnCpu(
           self: *MixtureOfExperts,
           hidden_states: mlx.Array,
       ) !RoutingResult {
           // Compute gate logits: [batch, seq, hidden] @ [hidden, num_experts]
           const logits = try mlx.matmul(
               hidden_states,
               self.gate_weight,
               .{ .transpose_b = true }
           );
           
           // Softmax over experts dimension (-1)
           const probs = try mlx.softmax(logits, .{ .axis = -1 });
           
           // Top-k selection
           const topk_result = try mlx.topk(
               probs,
               @intCast(self.config.top_k),
               .{ .axis = -1 }
           );
           
           return .{
               .indices = topk_result.indices,
               .weights = topk_result.values,
           };
       }
       ```
    
    2. Update route() to use kernel if available, else CPU:
