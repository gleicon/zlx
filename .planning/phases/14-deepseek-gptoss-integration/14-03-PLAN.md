## DeepSeek GGUF Requirements

**Model Source:** TheBloke/deepseek-coder-v2-lite-gguf or unsloth/DeepSeek-Coder-V2-Lite-GGUF
**Format:** GGUF v3 with Q4_K_M or Q5_K_M quantization
**File Size:** ~4-5GB for 16B active parameters

**GGUF Naming Convention:**
```
deepseek-coder-v2-lite-Q4_K_M.gguf
├── Q4_K_M = 4-bit K-quants with medium mix
├── Supports MoE routing natively
├── KV cache compatible with TurboQuant
└── Metadata contains tokenizer config
```

## llama.cpp DeepSeek Support

**Status:** llama.cpp has native DeepSeek V2 support as of b4000+
**Key Features:**
- MLA (Multi-head Latent Attention) via ggml_flash_attn
- MoE routing with 64 experts, top-6
- Q4/Q5/Q6/K-quant formats
- Metal GPU offload for all layers

## Sampling Strategy

llama.cpp samplers (in order):
1. Top-K (keep k highest probability tokens)
2. Top-P/nucleus (cumulative probability cutoff)
3. Temperature (softmax scaling)
4. Greedy (argmax when temperature ≈ 0)

```zig
// Sample chain setup
var sampler_params = llama_c.llama_sampler_chain_default_params();
var chain = llama_c.llama_sampler_chain_init(sampler_params);
llama_c.llama_sampler_chain_add(chain, llama_c.llama_sampler_init_top_k(params.top_k));
llama_c.llama_sampler_chain_add(chain, llama_c.llama_sampler_init_top_p(params.top_p, 1));
llama_c.llama_sampler_chain_add(chain, llama_c.llama_sampler_init_temp(params.temperature));
```

    const std = @import("std");
    const llama_c = @import("llama_c.zig");
    const backend = @import("backend.zig");
    
    pub const LlamaBackend = struct {
        model: *llama_c.llama_model,
        ctx: *llama_c.llama_context,
        allocator: std.mem.Allocator,
        vocab_size: u32,
        eos_token: llama_c.llama_token,
        
        pub fn init(allocator: std.mem.Allocator, model_path: []const u8) !LlamaBackend {
            // 1. Set up model parameters
            var model_params = llama_c.llama_model_default_params();
            model_params.n_gpu_layers = 1000; // Offload all layers to Metal
            
            // 2. Load model from GGUF file
            const model = llama_c.llama_model_load_from_file(
                model_path.ptr,
                model_params,
            ) orelse return error.ModelLoadFailed;
            
            // 3. Set up context parameters
            var ctx_params = llama_c.llama_context_default_params();
            ctx_params.n_ctx = 8192; // Default context size
            ctx_params.n_batch = 512;
            ctx_params.n_threads = @intCast(std.Thread.getCpuCount() catch 4);
            ctx_params.n_threads_batch = ctx_params.n_threads;
            
            // 4. Create context
            const ctx = llama_c.llama_new_context_with_model(model, ctx_params) orelse {
                llama_c.llama_model_free(model);
                return error.ContextCreationFailed;
            };
            
            // 5. Get metadata
            const vocab_size = @as(u32, @intCast(llama_c.llama_n_vocab(model)));
            const eos_token = llama_c.llama_token_eos(model);
            
            std.log.info("Loaded model: {s}", .{model_path});
            std.log.info("Vocab size: {d}, EOS token: {d}", .{vocab_size, eos_token});
            
            return LlamaBackend{
                .model = model,
                .ctx = ctx,
                .allocator = allocator,
                .vocab_size = vocab_size,
                .eos_token = eos_token,
            };
        }
        
        pub fn deinit(self: *LlamaBackend) void {
            llama_c.llama_free(self.ctx);
            llama_c.llama_model_free(self.model);
        }
        
        pub fn vocabSize(self: LlamaBackend) u32 {
            return self.vocab_size;
        }
        
        pub fn eosToken(self: LlamaBackend) llama_c.llama_token {
            return self.eos_token;
        }
    };
    ```

    // Add to LlamaBackend in llama_cpp.zig
    
    pub fn tokenize(self: LlamaBackend, text: []const u8, allocator: std.mem.Allocator) ![]llama_c.llama_token {
        // Allocate buffer for tokens (text.len is upper bound)
        const max_tokens = text.len + 8; // Extra for BOS/EOS
        var tokens = try allocator.alloc(llama_c.llama_token, max_tokens);
        errdefer allocator.free(tokens);
        
        // Tokenize
        const n_tokens = llama_c.llama_tokenize(
            self.model,
            text.ptr,
            @intCast(text.len),
            tokens.ptr,
            @intCast(max_tokens),
            true,  // add_special (BOS)
            false, // parse_special
        );
        
        if (n_tokens < 0) {
            return error.TokenizationFailed;
        }
        
        // Resize to actual token count
        return try allocator.realloc(tokens, @intCast(n_tokens));
    }
    
    pub const GenerationResult = struct {
        tokens_generated: u32 = 0,
        finish_reason: []const u8 = "length",
        token_iterator: TokenIterator,
    };
    
    pub const TokenIterator = struct {
        backend: *LlamaBackend,
        sampler: *llama_c.llama_sampler,
        max_tokens: u32,
        tokens_generated: u32 = 0,
        
        pub fn next(self: *TokenIterator) ?llama_c.llama_token {
            if (self.tokens_generated >= self.max_tokens) {
                return null;
            }
            
            // Get logits from last decode
            const logits = llama_c.llama_get_logits(self.backend.ctx);
            const n_vocab = @as(usize, @intCast(llama_c.llama_n_vocab(self.backend.model)));
            
            // Sample next token
            const token = llama_c.llama_sample_token(
                self.backend.ctx,
                self.sampler,
                logits,
                n_vocab,
            );
            
            self.tokens_generated += 1;
            
            // Check for EOS
            if (token == self.backend.eos_token) {
                return null;
            }
            
            // Decode the new token for next iteration
            var batch = llama_c.llama_batch_init(1, 0, 1);
            defer llama_c.llama_batch_free(batch);
            
            batch.token[0] = token;
            batch.pos[0] = @intCast(llama_c.llama_get_kv_cache_token_count(self.backend.ctx));
            batch.n_tokens = 1;
            
            if (llama_c.llama_decode(self.backend.ctx, batch) != 0) {
                return null; // Decode failed
            }
            
            return token;
        }
    };
    
    pub fn generate(
        self: *LlamaBackend,
        tokens: []const llama_c.llama_token,
        params: backend.GenerationParams,
    ) !GenerationResult {
        // 1. Create sampler chain
        var sampler_params = llama_c.llama_sampler_chain_default_params();
        const sampler = llama_c.llama_sampler_chain_init(sampler_params);
        errdefer llama_c.llama_sampler_free(sampler);
        
        // Add samplers in order: top_k → top_p → temperature
        if (params.top_k > 0) {
            llama_c.llama_sampler_chain_add(
                sampler,
                llama_c.llama_sampler_init_top_k(@intCast(params.top_k)),
            );
        }
        if (params.top_p < 1.0) {
            llama_c.llama_sampler_chain_add(
                sampler,
                llama_c.llama_sampler_init_top_p(params.top_p, 1),
            );
        }
        if (params.temperature > 0) {
            llama_c.llama_sampler_chain_add(
                sampler,
                llama_c.llama_sampler_init_temp(params.temperature),
            );
        }
        
        // 2. Initial decode of prompt tokens
        var batch = llama_c.llama_batch_init(@intCast(tokens.len), 0, 1);
        defer llama_c.llama_batch_free(batch);
        
        for (tokens, 0..) |token, i| {
            batch.token[i] = token;
            batch.pos[i] = @intCast(i);
            batch.seq_id[i] = 0;
        }
        batch.n_tokens = @intCast(tokens.len);
        
        if (llama_c.llama_decode(self.ctx, batch) != 0) {
            return error.InitialDecodeFailed;
        }
        
        // 3. Create iterator for token generation
        var iterator = TokenIterator{
            .backend = self,
            .sampler = sampler,
            .max_tokens = params.max_tokens,
            .tokens_generated = 0,
        };
        
        return GenerationResult{
            .token_iterator = iterator,
            .tokens_generated = 0,
            .finish_reason = "length",
        };
    }
    ```
    
    Notes:
    - llama_batch management is crucial for correct token positioning
    - KV cache grows automatically with each decode
    - Sampler owns sampling strategy; must be freed after generation
    - Check return codes from llama_decode()
