## Current Generator Architecture

From src/inference/generator.zig (Phase 13):
```zig

pub const TransformerUnion = union(enum) {
    qwen: qwen.Transformer,
    deepseek: deepseek.DeepSeekTransformer,
    gpt_oss: gpt_oss.GptOssTransformer,
};
```

**Problem:** Hardcoded MLX.zig dependencies throughout.

## Target Architecture

```zig
pub const Generator = struct {
    allocator: std.mem.Allocator,
    model_manager: *ModelManager,
    backend: backends.Backend,  // ← Unified backend interface
    tokenizer: tokenizer.Tokenizer,
    cache: ?*prompt_cache.PromptCache,
    compression: ?*compression.CompressionEngine,
};
```

## Backend Selection Flow

1. User requests model (e.g., `zlx --model deepseek-coder-v2-lite`)
2. Registry lookup → ModelInfo with architecture
3. Factory.createBackend(arch) → Backend union
4. Generator stores backend and uses it for all inference
5. Tokenization, generation, KV cache all go through backend

## TurboQuant Integration

TurboQuant needs KV cache access for compression:

```zig
// Current (MLX.zig specific):
const kv_cache = self.transformer.kv_cache;
compression.compress(kv_cache);

// Target (Backend agnostic):
const kv_handle = self.backend.getKvCache();
compression.compress(kv_handle);
```

Need to add to Backend interface:
```zig
pub fn getKvCache(self: Backend) KvCacheHandle;
pub fn applyCompression(self: Backend, compression: CompressionParams) void;
```

## Migration Strategy

Phase 1: Add Backend field alongside existing fields (parallel)
Phase 2: Implement Backend methods that delegate to MLX/llama
Phase 3: Switch Generator to use Backend exclusively
Phase 4: Remove old MLX-specific code

    const std = @import("std");
    const backends = @import("../backends/mod.zig");

    // ... other imports
    
    pub const Generator = struct {
        allocator: std.mem.Allocator,
        model_manager: *ModelManager,
        backend: backends.Backend,
        tokenizer: tokenizer.Tokenizer,
        cache: ?*prompt_cache.PromptCache,
        compression: ?*compression.CompressionEngine,
        metrics: ?*metrics.MetricsCollector,
        
        // Legacy fields (remove in Phase 4)
        mlx_context: ?mlx.Context = null,
        transformer: ?TransformerUnion = null,
        
        pub const InitOptions = struct {
            model_path: []const u8,
            model_architecture: registry.ModelArchitecture,
            backend_preference: factory.BackendPreference = .auto,
            enable_turboquant: bool = false,
            turboquant_bits: u4 = 4,
            enable_cache: bool = true,
            cache_dir: ?[]const u8 = null,
        };
        
        pub fn init(allocator: std.mem.Allocator, options: InitOptions) !Generator {
            // 1. Create backend via factory (auto-selects based on architecture)
            const backend = try factory.createBackend(
                allocator,
                options.model_path,
                options.model_architecture,
                options.backend_preference,
            );
            errdefer backend.deinit(allocator);
            
            std.log.info("Initialized {s} backend for {s}", .{
                @tagName(backend.getType()),
                @tagName(options.model_architecture),
            });
            
            // 2. Load tokenizer (backend-agnostic path)
            const tokenizer = try loadTokenizer(allocator, options.model_path);
            errdefer tokenizer.deinit();
            
            // 3. Initialize prompt cache if enabled
            var cache: ?*prompt_cache.PromptCache = null;
            if (options.enable_cache) {
                cache = try prompt_cache.init(allocator, options.cache_dir orelse DEFAULT_CACHE_DIR);
            }
            
            // 4. Initialize compression if TurboQuant enabled
            var compression: ?*compression.CompressionEngine = null;
            if (options.enable_turboquant) {
                compression = try compression.init(allocator, .{
                    .bits = options.turboquant_bits,
                    .adaptive_layers = 4,
                });
                
                // Apply compression to backend KV cache
                try backend.applyCompression(.{
                    .enabled = true,
                    .bits = options.turboquant_bits,
                    .adaptive_layers = 4,
                });
            }
            
            return Generator{
                .allocator = allocator,
                .model_manager = undefined, // Set by caller
                .backend = backend,
                .tokenizer = tokenizer,
                .cache = cache,
                .compression = compression,
                .metrics = null,
                // Legacy: null until migration complete
                .mlx_context = null,
                .transformer = null,
            };
        }
        
        pub fn deinit(self: *Generator) void {
            self.backend.deinit(self.allocator);
            self.tokenizer.deinit();
            if (self.cache) |cache| cache.deinit();
            if (self.compression) |comp| comp.deinit();
            if (self.metrics) |m| m.deinit();
            
            // Legacy cleanup
            if (self.mlx_context) |ctx| mlx.freeContext(ctx);
        }
        
        pub fn generate(
            self: *Generator,
            request: GenerationRequest,
        ) !GenerationResult {
            // 1. Tokenize prompt using backend
            const prompt_tokens = try self.backend.tokenize(
                request.prompt,
                self.allocator,
            );
            defer self.allocator.free(prompt_tokens);
            
            // 2. Check prompt cache
            var cache_hit = false;
            var cached_tokens: ?[]u32 = null;
            if (self.cache) |cache| {
                cached_tokens = try cache.get(request.prompt, self.allocator);
                cache_hit = cached_tokens != null;
            }
            
            // 3. Generate using backend
            const params = backends.GenerationParams{
                .temperature = request.temperature,
                .top_p = request.top_p,
                .top_k = request.top_k,
                .max_tokens = request.max_tokens,
                .seed = request.seed,
                .stop_sequences = request.stop_sequences,
                .presence_penalty = request.presence_penalty,
                .frequency_penalty = request.frequency_penalty,
            };
            
            const gen_result = try self.backend.generate(
                cached_tokens orelse prompt_tokens,
                params,
            );
            
            // 4. Update cache with new tokens
            if (self.cache) |cache| {
                try cache.put(request.prompt, gen_result.tokens_generated);
            }
            
            return gen_result;
        }
        
        // Legacy compatibility methods (to be removed)
        pub fn generateLegacy(self: *Generator, request: GenerationRequest) !GenerationResult {
            // Fall back to old MLX-specific implementation
            // Only used during migration period
            _ = self;
            _ = request;
            return error.NotImplemented;
        }
    };

    fn createGeneratorForRequest(
        allocator: std.mem.Allocator,
        model_name: []const u8,
        config: ServerConfig,
    ) !generator.Generator {
        // 1. Look up model in registry
        const model_info = registry.getKnownModel(model_name) orelse {
            // Try as direct path
            return createGeneratorFromPath(allocator, model_name, config);
        };
        
        // 2. Resolve model path
        const model_path = try resolveModelPath(allocator, model_info);
        defer allocator.free(model_path);
        
        // 3. Determine backend preference
        const backend_pref = if (config.force_backend) |b|
            b
        else
            factory.detectBackendType(model_info.architecture);
        
        std.log.info("Creating generator for {s} with {s} backend", .{
            model_name,
            @tagName(backend_pref),
        });
        
        // 4. Create generator with backend
        return try generator.Generator.init(allocator, .{
            .model_path = model_path,
            .model_architecture = model_info.architecture,
            .backend_preference = backend_pref,
            .enable_turboquant = config.turboquant_enabled,
