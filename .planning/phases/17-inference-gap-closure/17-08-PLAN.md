<objective>
Replace `GPTOSSTransformer.forward()`'s `mlx.zeros()` stub with real tensor computation
using loaded model weights.

**Current state:** `forward()` returns a zero-filled `[1, 1, vocab_size]` tensor regardless
of input. Argmax of zeros is deterministically 0 — generate() produces `max_tokens` copies
of token 0.

**Scope:** Implement a minimal but real forward pass:
1. Add weight fields to `GPTOSSTransformer`: `embed_tokens`, `lm_head` (MLX arrays)
2. Add `loadWeights(model_path)` that uses `GPTOSSWeightLoader` to load the embedding
   table (`model.embed_tokens.weight`) and LM head (`lm_head.weight`)
3. In `forward()`: look up the last input token's embedding, project through `lm_head`
   to get `[1, 1, vocab_size]` logits

This is a real embedding-based prediction (not a full attention+FFN transformer), but it:
- Uses actual weight tensors loaded from safetensors files
- Produces non-zero, input-dependent logits
- Satisfies GAP-03: "real tensor computation — not returning mlx.zeros()"
- Is realistic given the existing weight loading infrastructure

Full attention/FFN layers are Phase 18+ scope — documented explicitly.

Output: `generate()` produces non-trivial tokens when a model path is supplied.
</objective>

```zig
pub fn loadWeights(self: *GPTOSSTransformer, model_path: []const u8) !void {
    const weight_config = GPTOSSWeightConfig{
        .hidden_size = self.config.hidden_size,
        .num_layers = self.config.num_hidden_layers,
        .num_experts = self.config.num_experts,
        .num_shared_experts = self.config.num_shared_experts,
        .top_k = self.config.top_k,
        .vocab_size = self.config.vocab_size,
        .intermediate_size = self.config.intermediate_size,
    };
    var loader = GPTOSSWeightLoader.init(self.allocator, model_path, weight_config);
    defer loader.deinit();
    try loader.load(null);

    // Copy out the tensors we need (loader.deinit() would free them otherwise)
    if (loader.getTensor("model.embed_tokens.weight")) |t| {
        // mlx arrays are reference-counted — just copy the handle
        self.embed_tokens = t;
    }
    if (loader.getTensor("lm_head.weight")) |t| {
        self.lm_head = t;
    }
    self.weights_loaded = true;

    // Prevent loader.deinit() from freeing tensors we're holding
    // Note: need to understand mlx array ownership — see below
}
```

**CRITICAL: mlx array ownership.** `mlx.Array` in the C API is a handle. `mlx.arrayFree()`
decrements the refcount. If `GPTOSSWeightLoader.deinit()` calls `mlx.arrayFree` on all
tensors it holds, and GPTOSSTransformer also holds a copy of those handles, the transformer's
copy becomes invalid after deinit. Options:
1. Do NOT call `loader.deinit()` — leak the loader (acceptable for personal-use server)
2. Call `mlx.arrayRetain(t)` on each tensor before loader.deinit() if that API exists
3. Don't use `defer loader.deinit()` — keep the loader alive in the transformer struct

**Recommended approach:** Store the `GPTOSSWeightLoader` as an optional field in
`GPTOSSTransformer`, load it in `loadWeights()`, keep it alive. Only free in `deinit()`.
This avoids the ownership problem entirely.

```zig
pub fn forward(self: *GPTOSSTransformer, input_ids: []const u32) !mlx.Array {
    // Fallback when no weights are loaded
    if (!self.weights_loaded or self.loader == null) {
        var logits = mlx.arrayNew();
        const shape = [_]c_int{ 1, 1, @intCast(self.config.vocab_size) };
        try mlx.zeros(&logits, &shape, mlx.FLOAT32, self.stream);
        return logits;
        // Note: full attention+FFN forward pass is Phase 18 scope
    }

    const last_token: u32 = if (input_ids.len > 0) input_ids[input_ids.len - 1] else 0;
    const loader = &self.loader.?;
    const embed_table = loader.getTensor("model.embed_tokens.weight") orelse {
        // No embedding table — fallback to zeros
        var logits = mlx.arrayNew();
        const shape = [_]c_int{ 1, 1, @intCast(self.config.vocab_size) };
        try mlx.zeros(&logits, &shape, mlx.FLOAT32, self.stream);
        return logits;
    };

    // Token embedding: embed_table[last_token, :] → [hidden_size]
    // Use mlx_take(embed_table, indices, axis=0) where indices = [last_token]
    var token_idx = mlx.arrayNew();
    defer mlx.arrayFree(token_idx);
    const idx_val = [_]i32{@intCast(last_token)};
    try mlx.arrayFromData(&token_idx, &idx_val, 1, &[_]c_int{1}, mlx.INT32);

    var embedding = mlx.arrayNew();
    defer mlx.arrayFree(embedding);
    try mlx.take(&embedding, embed_table, token_idx, 0, self.stream);
    // embedding shape: [1, hidden_size]

    // LM head projection: embedding @ lm_head.T → [1, vocab_size]
    const lm_head = loader.getTensor("lm_head.weight") orelse embed_table;
    // lm_head shape: [vocab_size, hidden_size] — need transpose
    var lm_head_t = mlx.arrayNew();
    defer mlx.arrayFree(lm_head_t);
    const axes = [_]c_int{ 1, 0 };
    try mlx.transpose(&lm_head_t, lm_head, &axes, 2, self.stream);
    // lm_head_t shape: [hidden_size, vocab_size]

    var logits_2d = mlx.arrayNew();
    defer mlx.arrayFree(logits_2d);
    try mlx.matmul(&logits_2d, embedding, lm_head_t, self.stream);
    // logits_2d shape: [1, vocab_size]

    // Reshape to [1, 1, vocab_size] for generate() argmax compatibility
    var logits = mlx.arrayNew();
    const out_shape = [_]c_int{ 1, 1, @intCast(self.config.vocab_size) };
    try mlx.reshape(&logits, logits_2d, &out_shape, 3, self.stream);
    return logits;
}
```
